<script>
/* 1. БАЗА ДАННЫХ И СОСТОЯНИЯ */
let db = [];
let state = {
mangaId: null,
chapterIndex: null
};

// Стартовые данные для примера (если хранилище пустое)
const sampleData = [{
id: 1700000000000,
title: "Добро пожаловать в MangaWeb",
cover: "https://images.unsplash.com/photo-1618331835717-801e976710b2?auto=format&fit=crop&q=80&w=500",
desc: "Это тестовая манга. Нажми «Новая манга» сверху, чтобы создать свою собственную историю, и добавляй главы!",
chapters: []
}];

/* 2. ИНИЦИАЛИЗАЦИЯ (При загрузке страницы) */
window.onload = function() {
// Загружаем данные из памяти телефона
const saved = localStorage.getItem('MangaProDB');
if (saved) {
try {
db = JSON.parse(saved);
} catch(e) {
db = sampleData; // Защита от поломки JSON
}
} else {
db = sampleData;
}
saveDB();
renderCatalog(db);
};

function saveDB() {
localStorage.setItem('MangaProDB', JSON.stringify(db));
}

/* 3. УПРАВЛЕНИЕ ВИДАМИ (ЭКРАНАМИ) */
function showView(viewId) {
// Прячем все экраны
document.getElementById('view-catalog').classList.add('hidden');
document.getElementById('view-details').classList.add('hidden');
document.getElementById('view-reader').classList.add('hidden');
document.getElementById('main-nav').classList.add('hidden');

// Показываем нужный
document.getElementById(viewId).classList.remove('hidden');

// Показываем навбар только на главной и деталях
if(viewId !== 'view-reader') {
document.getElementById('main-nav').classList.remove('hidden');
}

// Скролл вверх при смене экрана
window.scrollTo({ top: 0, behavior: 'instant' });
}

function goHome() {
state.mangaId = null;
state.chapterIndex = null;
document.getElementById('search-input').value = '';
renderCatalog(db);
showView('view-catalog');
}

/* 4. ОТРИСОВКА КАТАЛОГА */
function renderCatalog(dataToRender) {
const grid = document.getElementById('manga-grid');
const emptyState = document.getElementById('empty-catalog');
document.getElementById('manga-count').textContent = dataToRender.length + ' тайтлов';

grid.innerHTML = '';

if (dataToRender.length === 0) {
emptyState.classList.remove('hidden');
} else {
emptyState.classList.add('hidden');
dataToRender.forEach(manga => {
// Ищем последнюю главу для отображения бейджа
let lastChap = manga.chapters.length > 0
? manga.chapters[manga.chapters.length - 1].title
: "Нет глав";

grid.innerHTML += &lt;div class="manga-card cursor-pointer bg-slate-900 rounded-2xl overflow-hidden border border-slate-800 flex flex-col h-full relative" onclick="openDetails(${manga.id})">
<div class="relative pt-[140%] bg-slate-800">
<img src="${manga.cover}" alt="cover" class="absolute top-0 left-0 w-full h-full object-cover" onerror="this.src='https://via.placeholder.com/400x600/1e293b/94a3b8?text=Нет+обложки'"&gt; &lt;div class="absolute inset-0 bg-gradient-to-t from-slate-950 via-transparent to-transparent opacity-90"&gt;&lt;/div&gt; &lt;/div&gt; &lt;div class="absolute bottom-0 w-full p-4"&gt; &lt;h3 class="font-bold text-white leading-tight mb-1 line-clamp-2"&gt;${manga.title}</h3>
<p class="text-purple-400 text-xs font-medium"><i class="fa-solid fa-layer-group"></i> ${lastChap}&lt;/p&gt; &lt;/div&gt; &lt;/div&gt;;
});
}
}

function searchManga() {
const q = document.getElementById('search-input').value.toLowerCase();
const filtered = db.filter(m => m.title.toLowerCase().includes(q));
renderCatalog(filtered);
}

/* 5. ОТРИСОВКА ДЕТАЛЕЙ */
window.openDetails = function(mangaId) {
const manga = db.find(m => m.id === mangaId);
if(!manga) return;

state.mangaId = mangaId;

document.getElementById('det-title').textContent = manga.title;
document.getElementById('det-desc').textContent = manga.desc || "Описание отсутствует.";
document.getElementById('det-cover').src = manga.cover;

const chapList = document.getElementById('det-chapters');
chapList.innerHTML = '';

if(manga.chapters.length === 0) {
chapList.innerHTML = '<div class="p-8 text-center text-slate-500 border-2 border-dashed border-slate-700 rounded-xl">Глав пока нет. Нажмите "Добавить главу".</div>';
} else {
// Выводим главы в обратном порядке (новые сверху)
const reversed = [...manga.chapters].reverse();

reversed.forEach((chap, revIndex) => {
// Оригинальный индекс в массиве нужен для читалки
const originalIndex = manga.chapters.length - 1 - revIndex;

chapList.innerHTML += &lt;div class="flex items-center justify-between bg-slate-900/80 hover:bg-slate-800 p-3 sm:p-4 rounded-xl border border-slate-800 transition-colors group"&gt; &lt;button class="flex-1 text-left font-bold text-white group-hover:text-purple-400 transition-colors" onclick="openReader(${originalIndex})">
${chap.title} &lt;/button&gt; &lt;div class="flex gap-2"&gt; &lt;button onclick="deleteChapter(${originalIndex})" class="w-10 h-10 flex items-center justify-center rounded-lg text-slate-500 hover:text-red-500 hover:bg-red-500/10 transition-colors" title="Удалить главу">
<i class="fa-solid fa-trash"></i>
</button>
<button onclick="openReader(${originalIndex})" class="w-10 h-10 flex items-center justify-center bg-purple-600/20 text-purple-400 rounded-lg hover:bg-purple-600 hover:text-white transition-colors"&gt; &lt;i class="fa-solid fa-play"&gt;&lt;/i&gt; &lt;/button&gt; &lt;/div&gt; &lt;/div&gt;;
});
}

showView('view-details');
}

/* 6. ЧИТАЛКА */
window.openReader = function(index) {
const manga = db.find(m => m.id === state.mangaId);
if(!manga || !manga.chapters[index]) return;

state.chapterIndex = index;
const chap = manga.chapters[index];

document.getElementById('read-manga-title').textContent = manga.title;
document.getElementById('read-chapter-title').textContent = chap.title;

// Настройка кнопок Следующая/Предыдущая
const btnPrev = document.getElementById('btn-prev-chap');
const btnNext = document.getElementById('btn-next-chap');

btnPrev.disabled = (index === 0); // Если это первая глава, отключаем "Предыдущую"
btnNext.disabled = (index === manga.chapters.length - 1); // Если последняя - "Следующую"

// Загрузка картинок
const imgContainer = document.getElementById('read-images');
imgContainer.innerHTML = '';

chap.images.forEach((url, i) => {
if(url.trim() === '') return;
imgContainer.innerHTML += &lt;img src="${url.trim()}"
alt="page-${i+1}" class="w-full object-contain m-0 block bg-black" onerror="this.onerror=null; this.src='https://via.placeholder.com/800x600/1e293b/ef4444?text=Ошибка+загрузки+страницы';"&gt;;
});

showView('view-reader');
}

window.closeReader = function() {
openDetails(state.mangaId);
}

window.changeChapter = function(direction) {
const manga = db.find(m => m.id === state.mangaId);
const newIndex = state.chapterIndex + direction;

if (newIndex >= 0 && newIndex < manga.chapters.length) {
openReader(newIndex);
}
}

/* 7. ФУНКЦИИ ДОБАВЛЕНИЯ И УДАЛЕНИЯ */
window.handleCreateManga = function(e) {
e.preventDefault();
const title = document.getElementById('inp-manga-title').value.trim();
const cover = document.getElementById('inp-manga-cover').value.trim();
const desc = document.getElementById('inp-manga-desc').value.trim();

const newManga = {
id: Date.now(),
title: title,
cover: cover,
desc: desc,
chapters: []
};

db.push(newManga);
saveDB();

e.target.reset(); // Очищаем форму
closeModal('modal-add-manga');
showNotification('Манга успешно создана!');
openDetails(newManga.id); // Сразу переходим в нее
}

window.handleCreateChapter = function(e) {
e.preventDefault();
const title = document.getElementById('inp-chap-title').value.trim();
const rawImages = document.getElementById('inp-chap-images').value;

// Разбиваем текст на строки и убираем пустые
const images = rawImages.split('\n').map(u => u.trim()).filter(u => u !== '');

const manga = db.find(m => m.id === state.mangaId);
manga.chapters.push({
title: title,
images: images
});

saveDB();
e.target.reset();
closeModal('modal-add-chapter');
showNotification('Глава опубликована!');
openDetails(state.mangaId); // Обновляем список глав на экране
}

window.deleteCurrentManga = function() {
if(confirm("Вы уверены, что хотите удалить эту мангу? Это действие нельзя отменить.")) {
db = db.filter(m => m.id !== state.mangaId);
saveDB();
showNotification('Манга удалена');
goHome();
}
}

window.deleteChapter = function(index) {
if(confirm("Удалить эту главу?")) {
const manga = db.find(m => m.id === state.mangaId);
manga.chapters.splice(index, 1);
saveDB();
showNotification('Глава удалена');
openDetails(state.mangaId);
}
}

/* 8. УПРАВЛЕНИЕ ОКНАМИ И ТЕМАМИ */
window.openModal = function(id) {
const m = document.getElementById(id);
m.classList.remove('hidden');
// Маленькая задержка для анимации появления
setTimeout(() => {
m.classList.remove('opacity-0');
m.querySelector('div').classList.remove('scale-95');
}, 10);
}

window.closeModal = function(id) {
const m = document.getElementById(id);
m.classList.add('opacity-0');
m.querySelector('div').classList.add('scale-95');
setTimeout(() => { m.classList.add('hidden'); }, 200); // Ждем конец анимации
}

window.setTheme = function(themeClass) {
const reader = document.getElementById('view-reader');
const nav = document.getElementById('reader-navbar');
const title = document.getElementById('read-manga-title');

// Сбрасываем старые темы
reader.classList.remove('theme-dark', 'theme-gray', 'theme-light');
nav.classList.remove('bg-slate-950/90', 'bg-white/90');
title.classList.remove('text-white', 'text-slate-900');

// Ставим новую
reader.classList.add(themeClass);

if (themeClass === 'theme-light') {
nav.classList.add('bg-white/90');
title.classList.add('text-slate-900');
} else {
nav.classList.add('bg-slate-950/90');
title.classList.add('text-white');
}
}

window.showNotification = function(msg) {
const t = document.getElementById('toast');
document.getElementById('toast-msg').textContent = msg;
t.classList.add('show');
setTimeout(() => { t.classList.remove('show'); }, 3000);
}

// Скрывать панель читалки при скролле вниз (погружение)
let lastScroll = 0;
window.addEventListener('scroll', () => {
if(document.getElementById('view-reader').classList.contains('hidden')) return;
const currentScroll = window.pageYOffset;
const nav = document.getElementById('reader-navbar');

if (currentScroll > lastScroll && currentScroll > 100) {
nav.style.transform = 'translateY(-100%)';
} else {
nav.style.transform = 'translateY(0)';
}
lastScroll = currentScroll;
});
